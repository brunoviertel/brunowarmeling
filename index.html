const API = "/api/clientes";

async function cadastrarCliente(dados) {
  const res = await fetch(API, {
    method: "POST",
    headers: {
      "Content-Type": "application/json",
    },
    body: JSON.stringify(dados),
  });
  if (!res.ok) {
    throw new Error("Falha");
  }
  return res.json();
}
